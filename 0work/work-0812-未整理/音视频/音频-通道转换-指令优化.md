```c
// 安全的标量版本，避免 NEON 相关问题
static INT32 priv_move_2SoundMode_1Sound_Safe(INT8* pDst, INT8* pSrc, UINT32 uDstLen, UINT32 uBytes)
{
    if(HIK_IS_NULL(pDst) || HIK_IS_NULL(pSrc) || (uBytes == 0) || (uDstLen == 0)) {
        return DSP_ERR_ADEC_NULL_PTR;
    }
    
    UINT32 uDstOffset = 0;
    UINT32 uSrcOffset = 0;
    UINT32 uStep = uBytes << 1; // 源步长 = 位宽 * 2
    
    // 逐字节拷贝，确保不越界
    for (uDstOffset = 0; uDstOffset < uDstLen && uSrcOffset < uDstLen * 2; uDstOffset += uBytes) {

            break;
        }
        
        HIK_MEM_CPY(pDst + uDstOffset, pSrc + uSrcOffset, uBytes);
        uSrcOffset += uStep;
    }
    
    return HIK_OK;
}
```
